cmake_minimum_required(VERSION 3.20)
project(BilliardsVision VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio)
add_library(billiards_core src/analyzer.cpp src/evaluation.cpp)
target_include_directories(billiards_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(billiards_core PUBLIC ${OpenCV_LIBS})
if(MSVC)
  target_compile_options(billiards_core PRIVATE /W4)
else()
  target_compile_options(billiards_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(billiards src/main.cpp)
target_link_libraries(billiards PRIVATE billiards_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(billiards_tests tests/tests.cpp)
  target_link_libraries(billiards_tests PRIVATE billiards_core)
  add_test(NAME algorithms COMMAND billiards_tests)
endif()
